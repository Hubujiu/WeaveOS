import { useEffect, useState } from 'react';
import { Modal } from '../../Modal';
import type { Application } from '../types';
import type { useApplicationOperation } from '../useApplicationOperation';

export function CreateApplication({ close, onDirty, operation }: { close: () => void; onDirty: (dirty: boolean) => void; operation: ReturnType<typeof useApplicationOperation<Application>> }) {
 const [name, setName] = useState(() => typeof operation.packet?.body.name === 'string' ? operation.packet.body.name : '');
 const [confirm, setConfirm] = useState(false);
 const busy = operation.phase === 'pending';
 const unconfirmed = operation.phase === 'unconfirmed';
 useEffect(() => { onDirty(name.length > 0 || unconfirmed); return () => onDirty(false); }, [name, onDirty, unconfirmed]);
 const requestClose = () => { if (busy) return; if (name.length || unconfirmed) setConfirm(true); else close(); };
 return <><Modal title="新建应用" onClose={requestClose} busy={busy}>
  <form className="app-create-form" onSubmit={e => { e.preventDefault(); operation.start('applications', { name }); }}>
   <label>应用名称<input autoFocus required name="application-name" value={name} disabled={busy || unconfirmed} onChange={e => setName(e.target.value)} /></label>
   {operation.message && <p role="alert">{operation.message}</p>}
   {unconfirmed ? <div className="dialog-actions"><button type="button" className="admin-button" onClick={operation.query}>核查操作</button><button type="button" className="admin-button primary" onClick={operation.retry}>使用同一操作重试</button></div>
    : <div className="dialog-actions"><button type="button" className="admin-button" disabled={busy} onClick={requestClose}>取消</button><button type="submit" className="admin-button primary" disabled={busy}>{busy ? '创建中…' : '创建应用'}</button></div>}
  </form>
 </Modal>{confirm && <Modal title="有未保存的修改" onClose={() => setConfirm(false)}><p>{unconfirmed ? '创建结果尚未确认。关闭后仍可在新建应用中核查原操作。' : '关闭将丢失未保存的应用名称。'}</p><div className="dialog-actions"><button className="admin-button" onClick={() => setConfirm(false)}>继续编辑</button><button className="admin-button primary" onClick={close}>放弃修改</button></div></Modal>}</>;
}
