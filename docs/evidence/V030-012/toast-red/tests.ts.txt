// Independent oracle: PRD V030-012.2/.3, AP-FR-10, B5a.4-7 and
// original Home358:18205/catalog327:2132. Mock only the HTTP boundary.
// These component fixtures are not real API acceptance.
const user = { id: '00000000-0000-4000-8000-000000000001', account: 'slice-owner' };
const application = {
  id: '00000000-0000-4000-8000-000000000002',
  name: '业务应用甲', ownerUserId: user.id, policyRevision: 1,
};
async function fixture(page: Page, create = true) {
  await page.route('**/api/v1/**', async route => {
    const url = new URL(route.request().url());
    const path = url.pathname.slice('/api/v1/'.length);
    const data = path === 'sessions/current' ? user
      : path === 'me/access' ? {
          user, bootstrapAdmin: false, personnelManage: false, identities: [],
          permissions: create ? [{ code: 'applications.create', name: '应用管理', category: 'system', appId: null }] : [],
          applications: [], // Deliberately empty old personnel catalogue.
        }
      : path === 'applications' ? { items: [application] }
      : path === 'applications/' + application.id ? application
      : path === 'applications/' + application.id + '/access' ? {
          appId: application.id, canEnter: true, policyRevision: 1,
          menus: [{ resourceKind: 'application', resourceId: application.id }],
        }
      : {};
    await route.fulfill({ status: 200, json: { code: 'OK', message: 'success', data, meta: null } });
  });
}

async function creation(page: Page) {
 await page.goto('/app/applications');
 await page.getByRole('button', { name: '新建应用', exact: true }).click();
 await page.getByRole('textbox', { name: '应用名称', exact: true }).fill('新业务应用');
 return page.getByRole('dialog', { name: '新建应用', exact: true });
}

test('V030-012 create waits for server acknowledgement and sends only the frozen DTO', async ({ page }) => {
 await fixture(page);
 let release!: () => void;
 const gate = new Promise<void>(resolve => { release = resolve; });
 const writes: { name: string; operationId: string }[] = [];
 await page.route('**/api/v1/applications', async route => {
  if (route.request().method() === 'GET') { await route.fulfill({ json: { code: 'OK', data: { items: [application] } } }); return; }
  writes.push(route.request().postDataJSON());
  await gate;
  await route.fulfill({ status: 201, json: { code: 'OK', data: { ...application, name: '新业务应用' } } });
 });
 const dialog = await creation(page);
 await dialog.getByRole('button', { name: '创建应用', exact: true }).click();
 await expect.poll(() => writes.length).toBe(1);
 expect(Object.keys(writes[0]).sort()).toEqual(['name', 'operationId']);
 expect(writes[0].name).toBe('新业务应用');
 expect(writes[0].operationId).toMatch(/^[0-9a-f]{8}-(?:[0-9a-f]{4}-){3}[0-9a-f]{12}$/);
 await expect(dialog.getByRole('button', { name: '创建中…', exact: true })).toBeDisabled();
 await expect(page.getByText('应用已创建', { exact: true })).toHaveCount(0);
 release();
 await expect(dialog).toHaveCount(0);
 await expect(page.getByRole('status').filter({ hasText: '应用已创建' })).toBeVisible();
});

test('V030-012 unconfirmed create recovers its original operation and never treats 404 as rollback', async ({ page }) => {
 await fixture(page);
 const writes: object[] = [];
 let queries = 0;
 await page.route('**/api/v1/applications', async route => {
  if (route.request().method() === 'GET') { await route.fulfill({ json: { code: 'OK', data: { items: [] } } }); return; }
  writes.push(route.request().postDataJSON());
  await route.fulfill({ status: 503, json: { code: 'APPLICATION_OPERATION_UNCONFIRMED', data: null } });
 });
 await page.route('**/api/v1/application-operations/*', async route => {
  queries++;
  expect(route.request().url().split('/').pop()).toBe((writes[0] as { operationId: string }).operationId);
  await route.fulfill(queries === 1 ? { status: 404, json: { code: 'APPLICATION_NOT_FOUND', data: null } } : { json: {
   code: 'OK', data: { operationId: (writes[0] as { operationId: string }).operationId, status: 'confirmed', httpStatus: 201, location: '/api/v1/applications/' + application.id, result: application },
  } });
 });
 const dialog = await creation(page);
 await dialog.getByRole('button', { name: '创建应用', exact: true }).click();
 await expect(dialog.getByRole('alert')).toContainText('操作结果尚未确认');
 await expect(dialog.getByRole('textbox', { name: '应用名称', exact: true })).toBeDisabled();
 await dialog.getByRole('button', { name: '核查操作', exact: true }).click();
 await expect(dialog.getByRole('alert')).toContainText('仍未确认');
 await expect(dialog.getByRole('alert')).not.toContainText('已回滚');
 await dialog.getByRole('button', { name: '核查操作', exact: true }).click();
 await expect(dialog).toHaveCount(0);
 await expect(page.getByRole('status').filter({ hasText: '应用已创建' })).toBeVisible();
 expect(writes).toHaveLength(1);
});

test('V030-012 retry after lost response preserves the operation key and exact payload', async ({ page }) => {
 await fixture(page);
 const writes: object[] = [];
 await page.route('**/api/v1/applications', async route => {

 await dialog.getByRole('button', { name: '使用同一操作重试', exact: true }).click();
 await expect(dialog).toHaveCount(0);
 expect(writes).toHaveLength(2);
 expect(writes[1]).toEqual(writes[0]);
});

test('V030-012 Home and application navigation retain one Shell and reauthorize repeated opens', async ({ page }) => {
 await fixture(page);
 let accesses = 0;
 page.on('request', request => { if (request.url().endsWith('/' + application.id + '/access')) accesses++; });
 await page.goto('/app/applications');
 const shell = await page.getByTestId('application-shell').elementHandle();
 for (let i = 0; i < 3; i++) {
  let before = accesses;
  await page.getByRole('main').getByRole('button', { name: application.name, exact: true }).click();
  await expect(page.getByRole('heading', { name: application.name, exact: true })).toBeVisible();
  expect(accesses).toBeGreaterThan(before);
  await page.getByRole('navigation', { name: '全局应用标签', exact: true }).getByRole('button', { name: '首页', exact: true }).click();
  before = accesses;
  await page.getByRole('navigation', { name: '全局应用标签', exact: true }).getByRole('button', { name: application.name, exact: true }).click();
  await expect(page.getByRole('heading', { name: application.name, exact: true })).toBeVisible();
  expect(accesses).toBeGreaterThan(before);
  await page.getByRole('navigation', { name: '全局应用标签', exact: true }).getByRole('button', { name: '全部应用', exact: true }).click();
  expect(await shell!.evaluate(node => node.isConnected)).toBe(true);
 }
 expect(accesses).toBeGreaterThanOrEqual(6);
});

test('V030-012 denial on same-key retry cannot erase an earlier unknown outcome', async ({ page }) => {
